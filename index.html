        // --- Click for Hearts Feature ---

        document.addEventListener('click', function(e) {

            heart.style.top = (e.clientY - 12) + 'px';

            

            // Random slight rotation

            const rot = Math.random() * 40 - 20;

            heart.style.transform = `rotate(${rot}deg)`;

            

            document.body.appendChild(heart);

            

            // Remove after animation

            setTimeout(() => {

                heart.remove();

            }, 2000);

        });


        // --- Confetti Feature ---

        function createConfetti() {

            const container = document.getElementById('confetti-container');

            const colors = ['#f43f5e', '#fb7185', '#fce7f3', '#ffffff', '#fda4af'];

            

            for (let i = 0; i < 50; i++) {

                const confetti = document.createElement('div');

                confetti.classList.add('confetti');

                

                // Randomize properties

                confetti.style.left = Math.random() * 100 + 'vw';

                confetti.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];

                confetti.style.opacity = Math.random() + 0.5;

                confetti.style.width = (Math.random() * 10 + 5) + 'px';

                confetti.style.height = (Math.random() * 5 + 5) + 'px';

                

                const animationDuration = Math.random() * 3 + 2;

                const animationDelay = Math.random() * 5;

                

                confetti.style.animationDuration = animationDuration + 's';

                confetti.style.animationDelay = animationDelay + 's';

                

                container.appendChild(confetti);

            }

        }

        // Run confetti on load

        window.onload = createConfetti;


        // --- Love Calculator Feature ---

        function calculateLove() {

            const container = document.getElementById('result-container');

            const bar = document.getElementById('progress-bar');

            const text = document.getElementById('calc-result');

            const msg = document.getElementById('calc-message');

            

            container.classList.remove('hidden');

            bar.style.width = '0%';

            text.innerText = '0%';

            msg.innerText = "Calculating chemistry...";

            

            let progress = 0;

            // Target is deliberately funny

            const target = 4200; 

            

            const interval = setInterval(() => {

                progress += Math.floor(Math.random() * 150) + 50;

                

                if (progress >= target) {

                    progress = target;

                    clearInterval(interval);

                    msg.innerText = "Error: Too much love to calculate! (Infinity %)";

                    msg.classList.add('text-brand-600', 'animate-pulse-fast');

                }

                

                // Visual bar stops at 100%, text goes higher

                bar.style.width = Math.min(progress, 100) + '%';

                text.innerText = progress + '%';

                

            }, 50);

        }


        // --- Memory Match Game Feature ---

        const images = [

            'image_b47b5c.jpg',

            'image_b47b62.jpg',

            'image_b47b80.jpg',

            'image_b47e27.jpg',

            'image_b47e45.jpg',

            'heart_icon' // special generic card

        ];

        

        let cardsArray = [];

        let hasFlippedCard = false;

        let lockBoard = false;

        let firstCard, secondCard;

        let moves = 0;

        let pairsFound = 0;


        function initGame() {

            const board = document.getElementById('memory-board');

            board.innerHTML = '';

            moves = 0;

            pairsFound = 0;

            updateStats();

            hasFlippedCard = false;

            lockBoard = false;

            firstCard = null;

            secondCard = null;

            

            // Create pairs

            cardsArray = [...images, ...images];

            // Shuffle

            cardsArray.sort(() => 0.5 - Math.random());

            

            // Generate HTML

            cardsArray.forEach((item, index) => {

                const card = document.createElement('div');

                card.classList.add('memory-card');

                card.dataset.framework = item;

                

                let frontContent = '';

                if(item === 'heart_icon') {

                    frontContent = `<div style="font-size: 3rem;">💖</div>`;

                } else {

                    frontContent = `<img src="${item}" alt="Memory">`;

                }


                card.innerHTML = `

                    <div class="front-face shadow-inner">

                        ${frontContent}

                    </div>

                    <div class="back-face shadow-lg">

                        <i class="fas fa-heart"></i>

                    </div>

                `;

                

                card.addEventListener('click', flipCard);

                board.appendChild(card);

            });

        }


        function flipCard() {

            if (lockBoard) return;

            if (this === firstCard) return;


            this.classList.add('flip');


            if (!hasFlippedCard) {

                // First click

                hasFlippedCard = true;

                firstCard = this;

                return;

            }


            // Second click

            secondCard = this;

            moves++;

            updateStats();

            checkForMatch();

        }


        function checkForMatch() {

            let isMatch = firstCard.dataset.framework === secondCard.dataset.framework;


            if (isMatch) {

                disableCards();

                pairsFound++;

                updateStats();

                

                if(pairsFound === images.length) {

                    setTimeout(() => {

                        alert("Yay! You found all our memories! Happy Anniversary! ❤️");

                        createConfetti(); // trigger more confetti

                    }, 500);

                }

            } else {

                unflipCards();

            }

        }


        function disableCards() {

            firstCard.removeEventListener('click', flipCard);

            secondCard.removeEventListener('click', flipCard);

            resetBoard();

        }


        function unflipCards() {

            lockBoard = true;

            setTimeout(() => {

                firstCard.classList.remove('flip');

                secondCard.classList.remove('flip');

                resetBoard();

            }, 1000);

        }


        function resetBoard() {

            [hasFlippedCard, lockBoard] = [false, false];

            [firstCard, secondCard] = [null, null];

        }


        function updateStats() {

            document.getElementById('moves-count').innerText = moves;

            document.getElementById('pairs-count').innerText = pairsFound;

        }


        // Initialize game on load

        document.addEventListener('DOMContentLoaded', initGame);
